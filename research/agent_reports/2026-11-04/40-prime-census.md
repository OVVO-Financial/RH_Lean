using Z = long long;
using E = std::pair<Z, Z>;

E mul(E x, E y) {
    return {x.first*y.first-x.second*y.second,
            x.first*y.second+x.second*y.first-x.second*y.second};
}
Z norm(E x) { return x.first*x.first-x.first*x.second+x.second*x.second; }
E conj(E x) { return {x.first-x.second, -x.second}; }
bool divides(E p, E x) {
    E v = mul(x, conj(p));
    return v.first % norm(p) == 0 && v.second % norm(p) == 0;
}
unsigned support(E z, const std::array<E, 7>& primes) {
    unsigned mask = 0;
    for (unsigned j = 0; j < primes.size(); ++j)
        if (divides(primes[j], z)) mask |= 1U << j;
    return mask;
}

            for (E i : ideals) {
                Z n = norm(i)*nb*nb*nb;
                if (n > 2*T) break;
                if (n <= T) continue;
                E v = mul(i, b3);
                if (norm(v) != n) return 4;
                unsigned mask = support(v, primes);
                if (mask != (support(i, primes) | support(b, primes))) return 5;
                ++counts[mask];
            }
        }
        for (unsigned mask = 0; mask < counts.size(); ++mask)
            if (counts[mask]) cube << T << ' ' << mask << ' ' << counts[mask] << '\n';
    }
    std::cout << "lattice_ideals=" << count << " limit=" << limit << '\n';
}
