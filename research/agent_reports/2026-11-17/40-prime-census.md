def require(label, condition):

@lru_cache(None)
def factors(n):
    out = {}
    d = 2
    while d*d <= n:
        while n%d == 0:
            out[d] = out.get(d,0)+1
            n //= d
        d += 1
    if n>1:
        out[n] = out.get(n,0)+1
    return out


def mobius(n):
    f = factors(n)
    return 0 if any(e>1 for e in f.values()) else (-1)**len(f)


def add(*polys):
    out = {}
    for poly in polys:
        for p,v in poly.items():
            out[p] = out.get(p,Q(0))+v
    return {p:v for p,v in out.items() if v}


def scale(poly, q):
    return {p:v*q for p,v in poly.items() if v*q}


def logpoly(n):
    return {p:Q(e) for p,e in factors(n).items()}


def minus(a,b):
    return add(a,scale(b,-1))


def serialize(poly):
    return [[p,str(v)] for p,v in sorted(poly.items())]

def ceildiv(a, b):
    return -((-a) // b)


def rat(n, d=1):
    assert d > 0
    return n * Q // d, ceildiv(n * Q, d)


def add(*args):
    return sum(a[0] for a in args), sum(a[1] for a in args)


def neg(a):
    return -a[1], -a[0]


def sub(a, b):
    return add(a, neg(b))


def scale(a, n, d=1):
    assert d > 0
    ends = (a[0] * n, a[1] * n)
    return min(ends) // d, ceildiv(max(ends), d)


def mul(a, b):
    ends = [x * y for x in a for y in b]
    return min(ends) // Q, ceildiv(max(ends), Q)


def square(a):
    lo = 0 if a[0] <= 0 <= a[1] else min(a[0]**2, a[1]**2)
    return lo // Q, ceildiv(max(a[0]**2, a[1]**2), Q)


def div(a, b):
    if b[1] < 0:
        return div(neg(a), neg(b))
    assert b[0] > 0
    return mul(a, (Q * Q // b[1], ceildiv(Q * Q, b[0])))


def overlap(a, b):
    return a[0] <= b[1] and b[0] <= a[1]


def contains(a, n):
    return a[0] <= n * Q <= a[1]


def display(a):
    # Outward decimal endpoints at 12 places.
    base = Q // 10**12
    def fmt(n):
        sign = '-' if n < 0 else ''
        n = abs(n)
        return f'{sign}{n // 10**12}.{n % 10**12:012d}'
    return [fmt(a[0] // base), fmt(ceildiv(a[1], base))]


def log_unit(n, d):
    assert d <= n <= 2*d
    z = rat(n-d, n+d)
    z2 = mul(z, z)

    return add(scale(total, 2), (0, 1))


LOG2 = log_unit(2, 1)


def logn(n):
    e = n.bit_length()-1
    return add(log_unit(n, 1 << e), scale(LOG2, e))


def arith(limit):
    spf = list(range(limit+1))
    for p in range(2, isqrt(limit)+1):
        if spf[p] == p:
            for n in range(p*p, limit+1, p):
                if spf[n] == n:
                    spf[n] = p

    mu[1] = 1

        if p == n:
            logs[n] = logn(n)
            power = p
            while power <= limit:
                lam[power] = logs[n]
                power *= p

    out = [rat(0)]*(limit+1)

def mobius(n):
    parity = 0
    p = 2
    while p*p <= n:
        if n % p == 0:
            n //= p
            parity += 1
            if n % p == 0:
                return 0
        p += 1
    if n > 1:
        parity += 1
    return -1 if parity % 2 else 1


@lru_cache(None)
def logarithm(n):
    return D(n).ln()


def band(lo, hi, m):
    total = 0
    left = lo+1
    while left <= min(hi, m):
        quotient = m//left
        right = min(hi, m//quotient)
        total += quotient*(right-left+1)
        left = right+1
    return total
