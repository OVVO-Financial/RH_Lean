
~~~python
from itertools import product
from math import isqrt
import json
import os

checks = Counter()


def check(name, ok):
    assert ok, name
    checks[name] += 1


def ceildiv(a, b):
    return -((-a) // b)


def rat(n, d=1):
    assert d > 0
    return n * Q // d, ceildiv(n * Q, d)


def add(*args):
    return sum(a[0] for a in args), sum(a[1] for a in args)


def neg(a):
    return -a[1], -a[0]


def sub(a, b):
    return add(a, neg(b))


def scale(a, n, d=1):
    assert d > 0
    ends = (a[0] * n, a[1] * n)
    return min(ends) // d, ceildiv(max(ends), d)


def mul(a, b):
    ends = [x * y for x in a for y in b]
    return min(ends) // Q, ceildiv(max(ends), Q)


def square(a):
    lo = 0 if a[0] <= 0 <= a[1] else min(a[0]**2, a[1]**2)
    return lo // Q, ceildiv(max(a[0]**2, a[1]**2), Q)


def div(a, b):
    if b[1] < 0:
        return div(neg(a), neg(b))
    assert b[0] > 0
    return mul(a, (Q * Q // b[1], ceildiv(Q * Q, b[0])))


def overlap(a, b):
    return a[0] <= b[1] and b[0] <= a[1]


def contains(a, n):
    return a[0] <= n * Q <= a[1]

        sign = '-' if n < 0 else ''

    z2 = mul(z, z)

LOG2 = log_unit(2, 1)


def logn(n):
    e = n.bit_length()-1

def arith(limit):
    spf = list(range(limit+1))
    for p in range(2, isqrt(limit)+1):
        if spf[p] == p:
            for n in range(p*p, limit+1, p):
                if spf[n] == n:
                    spf[n] = p
    logs = [rat(0)]*(limit+1)
    mu = [0]*(limit+1)
    mu[1] = 1
    lam = [rat(0)]*(limit+1)
    for n in range(2, limit+1):
        p = spf[n]
        if p == n:
            logs[n] = logn(n)
            power = p
            while power <= limit:
                lam[power] = logs[n]
                power *= p
        else:
            logs[n] = add(logs[p], logs[n//p])
        mu[n] = 0 if (n//p) % p == 0 else -mu[n//p]
    psi = [rat(0)]*(limit+1)
    for n in range(1, limit+1):
        psi[n] = add(psi[n-1], lam[n])
    return mu, logs, psi


def floor_response(coeff, limit):
    inc = [rat(0)]*(limit+1)
    for k, c in coeff:
        if c == (0, 0):
            continue
        for n in range(k, limit+1, k):
            inc[n] = add(inc[n], c)
    out = [rat(0)]*(limit+1)
    for n in range(1, limit+1):
        out[n] = add(out[n-1], inc[n])
    return out


def band_floor(lo, hi, limit):
    inc = [0]*(limit+1)
    for k in range(lo+1, hi+1):
        for n in range(k, limit+1, k):
            inc[n] += 1
    out = [0]*(limit+1)
    for n in range(1, limit+1):
        out[n] = out[n-1]+inc[n]
    return out


def quotient_count(lo, hi, m):

                   mul(g[2][2], square(g[0][1]))))
~~~
