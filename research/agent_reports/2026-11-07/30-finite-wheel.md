## Reproducibility Appendix: Separate C++ Scalar Verifier

```cpp
#include <algorithm>
#include <iostream>
#include <numeric>
#include <sstream>
#include <string>
#include <vector>

struct Rational {
    long long n, d;
    Rational(long long a=0, long long b=1): n(a),d(b) {
        long long g=std::gcd(n,d); n/=g; d/=g;
    }
    Rational operator+(Rational b) const {return {n*b.d+b.n*d,d*b.d};}
    Rational operator-(Rational b) const {return {n*b.d-b.n*d,d*b.d};}
    Rational operator*(long long b) const {return {n*b,d};}
    std::string str() const {return std::to_string(n)+(d==1?"":"/"+std::to_string(d));}
};

int main() {
    const int L=512*512-1;
    std::vector<int> mu(L+1),least(L+1),primes,M(L+1),D(L+1);
    mu[1]=1;
    for(int n=2;n<=L;++n) {
        if(!least[n]) {least[n]=n;primes.push_back(n);mu[n]=-1;}
        for(int p:primes) {
            if(p>least[n] || 1LL*p*n>L) break;
            least[p*n]=p;
            mu[p*n]=(p==least[n]?0:-mu[n]);
        }
    }
    for(int n=1;n<=L;++n) {M[n]=M[n-1]+mu[n];D[n]=D[n-1]+mu[n]*mu[n];}
    for(int R=56;R<=512;++R) {
        int X=R*R-1,G=M[X]-M[R-1];
        Rational U,Uhi,H(D[X]-D[R-1]);
        long long E=0,Ehi=0;
        std::ostringstream labels;
        bool first=true;
        for(int q:primes) {
            if(q*q>=R) break;
            if(q==2) continue;
            int y=X/(q*q),d=M[y];
            U=U+Rational(d,q); E+=d*d;
            H=H+Rational(2*(D[y]-D[R-1]),q);
            if(R%q==0 && R<=q*q*q) {
                Uhi=Uhi+Rational(d,q);Ehi+=d*d;
                if(!first) labels<<",";
                labels<<q;first=false;
            }
        }
        Rational contribution=Uhi*(2*G)-Rational(2*Ehi);
        Rational full=Rational(G*G)+U*(2*G)-Rational(2*E)-H;
        std::cout<<R<<"|"<<labels.str()<<"|"<<G<<"|"<<Uhi.str()<<"|"
                 <<contribution.str()<<"|"<<full.str()<<"\n";
    }
}
```
