            n //= p
            sign = -sign
            if n % p == 0:
                return 0
        p += 1
    return -sign if n > 1 else sign

    return p >= 2 and all(p % d for d in range(2, isqrt(p) + 1))

    owners = [p for p in range(3, isqrt(R - 1) + 1, 2) if prime(p)]

from fractions import Fraction as F
from functools import cache
from math import gcd, lcm


class P:
    def __init__(self, value=0):
        self.d = ({k: F(v) for k, v in value.items() if v}
                  if isinstance(value, dict) else ({(): F(value)} if value else {}))

    def __add__(self, other):
        other = other if isinstance(other, P) else P(other)
        d = dict(self.d)
        for k, v in other.d.items():
            d[k] = d.get(k, F(0)) + v
        return P(d)

    __radd__ = __add__

    def __neg__(self):
        return P({k: -v for k, v in self.d.items()})

    def __sub__(self, other):
        return self + (-other if isinstance(other, P) else -P(other))

    def __rsub__(self, other):
        return P(other) - self

    def __mul__(self, other):
        other = other if isinstance(other, P) else P(other)
        d = {}
        for a, x in self.d.items():
            for b, y in other.d.items():
                key = tuple(sorted(a + b))
                d[key] = d.get(key, F(0)) + x * y
        return P(d)

    __rmul__ = __mul__

    def __truediv__(self, q):
        return self * (1 / F(q))

    def __eq__(self, other):
        return self.d == (other.d if isinstance(other, P) else P(other).d)


@cache
def factors(n):
    out, p = {}, 2
    while p * p <= n:
        while n % p == 0:
            out[p] = out.get(p, 0) + 1
            n //= p
        p += 1
    if n > 1:
        out[n] = out.get(n, 0) + 1
    return out


def mu(n):
    fs = factors(n)
    return 0 if any(e > 1 for e in fs.values()) else (-1) ** len(fs)


def log(n):
    return P({(p,): e for p, e in factors(n).items()})

    return sum((log(next(iter(factors(n))))
                for n in range(2, m + 1) if len(factors(n)) == 1), P())


def a(k, m):
    return F(m % k, k)

    return F(k - 1, 2 * k)

    return log(N) - sum((coeff[k] * a(k, m) for k in coeff), P())
