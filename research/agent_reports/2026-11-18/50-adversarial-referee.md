from collections import Counter
from fractions import Fraction as F
from math import gcd, isqrt, lcm
import hashlib
import json


COUNTS = Counter()


def check(label, proposition):
    assert proposition, label
    COUNTS[label] += 1


def factor(n):
    out = {}
    p = 2
    while p * p <= n:
        while n % p == 0:
            out[p] = out.get(p, 0) + 1
            n //= p
        p += 1
    if n > 1:
        out[n] = out.get(n, 0) + 1
    return out


def mu(n):
    fac = factor(n)
    return 0 if any(e > 1 for e in fac.values()) else (-1) ** len(fac)


def divs(n):
    return [d for d in range(1, n + 1) if n % d == 0]


def form(*entries):
    return {k: F(v) for k, v in entries if v}


def add(*terms):
    out = {}
    for term in terms:
        for k, v in term.items():
            out[k] = out.get(k, F(0)) + v
    return {k: v for k, v in out.items() if v}


def scale(term, weight):
    return {k: v * weight for k, v in term.items() if v * weight}


def logarithm(n):
    return {str(p): F(e) for p, e in factor(n).items()}


def mangoldt(n):
    fs = factor(n)
    return {str(next(iter(fs))): F(1)} if len(fs) == 1 else {}


def isum(terms):
    return add(*list(terms))


def conv(a, b, n):
    return sum((a[d] * b[n // d] for d in divs(n)), F(0))


def ln_small(x):
    z = (x - 1) / (x + 1)
    assert 0 <= z <= F(1, 3)
    s = 2 * sum((z ** (2 * j + 1) / (2 * j + 1) for j in range(48)), F(0))
    err = 2 * z ** 97 / (97 * (1 - z * z))
    return s, s + err


LN2 = ln_small(F(2))


def ln_integer(n):
    e = n.bit_length() - 1
    lo, hi = ln_small(F(n, 2 ** e))
    return lo + e * LN2[0], hi + e * LN2[1]


H64 = sum((F(1, k) for k in range(1, 65)), F(0))
GAMMA = (H64 - ln_integer(64)[1] - F(1, 64), H64 - ln_integer(64)[0])


def interval(x):
    lo = hi = F(0)

        lo += c * (a if c >= 0 else b)
        hi += c * (b if c >= 0 else a)
    return lo, hi


def abs_lower(x):
    a, b = interval(x)
    return 0 if a <= 0 <= b else min(abs(a), abs(b))


def rounded_lower(x, places=9):

    z = x.numerator * unit // x.denominator
    return f'{z // unit}.{z % unit:0{places}d}'
