import hashlib
import json
import math
from fractions import Fraction as F


def log_bounds(x, n=48):
    x = F(x)
    r = (x - 1) / (x + 1)
    assert 0 <= r < 1
    lo = 2 * sum((r ** (2 * k + 1) / (2 * k + 1) for k in range(n)), F(0))
    tail = 2 * r ** (2 * n + 1) / ((2 * n + 1) * (1 - r * r))
    return lo, lo + tail


def trial_lpf(n):
    if n < 2:
        return 0
    for p in range(2, math.isqrt(n) + 1):
        if n % p == 0:
            return p
    return n


def table(n):
    least = [0] * (n + 1)
    primes = []
    for p in range(2, n + 1):
        if least[p] == 0:
            primes.append(p)
            for k in range(p, n + 1, p):
                if least[k] == 0:
                    least[k] = p
    pi, density = [0] * (n + 1), [F(1)] * (n + 1)
    for k in range(2, n + 1):
        isprime = least[k] == k
        pi[k] = pi[k - 1] + isprime
        density[k] = density[k - 1] * (F(k - 1, k) if isprime else 1)
    return least, primes, pi, density


def constant_certificates():
    h = sum((F(1, j) for j in range(1, 33)), F(0))
    l2, u2 = log_bounds(2)
    l33, u33 = log_bounds(F(33, 32))
    l53, u53 = log_bounds(F(5, 3))
    l2011, _ = log_bounds(F(20, 11))
    # H_32-log(33) < gamma < H_32-log(32).
    assert h - (5 * u2 + u33) > u53
    assert h - 5 * l2 < l2011
    assert u53 < F(3, 5)
    assert F(5, 96) < F(1, 4)
    assert (F(3, 4) - F(3, 5)) / 2 > F(1, 20)
    for n in range(4, 101):
        bound = F((n + 1) ** 2, 5 * n * math.factorial(n))
        nxt = F((n + 2) ** 2, 5 * (n + 1) * math.factorial(n + 1))
        assert nxt < bound <= F(5, 96)
    return {"c_interval": "11/20 < exp(-gamma) < 3/5", "passed": True}


def wheel_control(R, primes, pi, density):
    X = R * R - 1
    terms = [(1, 1)]
    for p in primes:
        if p >= R:
            break
        terms += [(p * d, -mu) for d, mu in terms]
    physical = sum((mu * F(X % d, d) for d, mu in terms if d <= X), F(0))
    virtual = sum((mu * F(X, d) for d, mu in terms if d > X), F(0))
    floors = sum(mu * (X // d) for d, mu in terms)
    assert physical + virtual - X * density[R - 1] == -floors
    assert floors == 1 + pi[X] - pi[R - 1]
    return {"R": R, "physical": str(physical), "virtual": str(virtual), "floors": floors}


def probe(R, least, primes, pi, density, independent):
    X, h = R * R - 1, (R - 1) // 2
    a = math.isqrt(R - 1) + 1
    Q = [q for q in primes if R <= q < 2 * R]
    states = sorted({2, a - 1, a, h} | {p for p in primes if 2 <= p <= h})
    states = [z for z in states if 2 <= z <= h]
    W, H = density[h] - density[R - 1], pi[h]
    signed_improved, absolute_improved, rows = 0, 0, 0
    gap_min, sum_abs, sum_sq, sum_signed = None, F(0), F(0), F(0)
    vectors, minima = {}, []
    for q in Q:
        x, t = F(X, q), X // q
        r = x - t
        counts = [0] * (R + 1)
        for n in range(2, t + 1):
            counts[least[n]] += 1
        prefix = [0] * (R + 1)
        for k in range(1, R + 1):
            prefix[k] = prefix[k - 1] + counts[k]
        upper = sum(counts[p] for p in primes if R <= 2 * p and p < R)
        vals = {}
        for z in states:
            physical = prefix[z] + upper
            direct = physical - x * (1 - density[z] + W)
            rough = sum(1 for n in range(2, t + 1) if least[n] < n and least[n] > z)
            phi = 1 + sum(1 for n in range(2, t + 1) if least[n] > z)
            formula = (density[z] - W) * x + pi[z] - H - 1 - r - rough
            epsilon = t - phi - x * (1 - density[z])
            assert direct == formula == pi[t] - H - W * x + epsilon
            if z >= a:
                assert rough == 0
            if independent:
                slow_count = sum(1 for n in range(2, t + 1)
                                 if trial_lpf(n) <= z or R <= 2 * trial_lpf(n))
                assert physical == slow_count
            vals[z] = direct
            rows += 1
            if (R, q, z) in {(8, 11, 2), (10, 11, 2), (29, 29, 2)}:
                vectors[f"{R},{q},{z}"] = {"T": str(direct), "C": rough, "t": t}
        v_all = min(vals.values())
        v_sat = min(v for z, v in vals.items() if z >= a)
        e_all = min(abs(v) for v in vals.values())
        e_sat = min(abs(v) for z, v in vals.items() if z >= a)
        v_small = min(v for z, v in vals.items() if z < a)
        gap = v_small - v_sat
        if gap_min is None or gap < gap_min[0]:
            gap_min = (gap, q)
        signed_improved += v_all < v_sat
        absolute_improved += e_all < e_sat
        choice = min(z for z, v in vals.items() if abs(v) == e_all)
        sum_abs += e_all
        sum_sq += e_all * e_all
        sum_signed += vals[choice]
        minima.append((q, v_all, v_sat, e_all, e_sat))
    return {
        "R": R, "m": len(Q), "rows": rows,
        "signed_improved_fibres": signed_improved,
        "absolute_improved_fibres": absolute_improved,
        "small_gap": str(gap_min[0]), "small_gap_q": gap_min[1],
        "gap_times_logR_over_R_DIAGNOSTIC": float(gap_min[0]) * math.log(R) / R,
        "A1_times_logR2_over_R2_DIAGNOSTIC": float(sum_abs) * math.log(R) ** 2 / R ** 2,
        "A2_times_logR3_over_R3_DIAGNOSTIC": float(sum_sq) * math.log(R) ** 3 / R ** 3,
        "Eopt_times_logR2_over_R2_DIAGNOSTIC": float(sum_signed) * math.log(R) ** 2 / R ** 2,
        "witnesses": vectors,
    }
