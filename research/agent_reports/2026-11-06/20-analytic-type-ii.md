"""Exact source and scope checks for the Nov 6 CODIV frequency theorem."""

from array import array
from fractions import Fraction as F
from functools import lru_cache
from math import isqrt


def mobius_sieve(limit):
    mu = array("b", [1]) * (limit + 1)
    mu[0] = 0
    composite = bytearray(limit + 1)
    primes = []
    for p in range(2, limit + 1):
        if composite[p]:
            continue
        primes.append(p)
        for n in range(p, limit + 1, p):
            composite[n] = 1
            mu[n] = -mu[n]
        if p * p <= limit:
            for n in range(p * p, limit + 1, p * p):
                mu[n] = 0
    prefix = array("i", [0]) * (limit + 1)
    for n in range(1, limit + 1):
        prefix[n] = prefix[n - 1] + mu[n]
    return mu, prefix, primes


def divisor_inverse(limit):
    out = [0] * (limit + 1)
    out[1] = 1
    for d in range(1, limit + 1):
        for n in range(2 * d, limit + 1, d):
            out[n] -= out[d]
    return out


def clean(poly):
    return {p: v for p, v in poly.items() if v}


def amplitude_coefficients(root, prefix, primes):
    endpoint = root * root - 1
    out = {1: F(prefix[root - 1] - prefix[endpoint])}
    for p in primes:
        if p * p >= root:
            break
        if p != 2:
            out[p] = -F(prefix[endpoint // (p * p)], p)
    return clean(out)


def step_coefficients(root, mu, prefix, primes, include_birth=True):
    lower = root * root - 1
    upper = (root + 1) ** 2 - 1
    shell = prefix[upper] - prefix[lower]
    c = mu[root] - shell
    daughters = {}
    for p in primes:
        if p * p > root:
            break
        if p == 2:
            continue
        if p * p < root:
            b = prefix[upper // (p * p)] - prefix[lower // (p * p)]
        else:
            b = prefix[root + 2] if include_birth else 0
        daughters[p] = F(b, p)
    return c, daughters


@lru_cache(None)
def log_interval(ratio):
    assert 1 <= ratio <= 2
    t = (ratio - 1) / (ratio + 1)
    terms = 32
    lower = 2 * sum((t ** (2 * k + 1) / (2 * k + 1)
                     for k in range(terms)), F(0))
    error = 2 * t ** (2 * terms + 1) / (
        (2 * terms + 1) * (1 - t * t))
    return lower, lower + error


def triangular_kernel_interval(p, q):
    ratio = F(max(p, q), min(p, q))
    if ratio >= 2:
        return F(0), F(0)
    if ratio == 1:
        return F(1), F(1)
    lo, hi = log_interval(ratio)
    log2lo, log2hi = log_interval(F(2))
    return 1 - hi / log2lo, 1 - lo / log2hi


def weighted_interval(poly, kernel):
    lo = hi = F(0)
    for p, cp in poly.items():
        for q, cq in poly.items():
            kl, kh = kernel(p, q)
            factor = cp * cq
            if factor >= 0:
                lo += factor * kl
                hi += factor * kh
            else:
                lo += factor * kh
                hi += factor * kl
    return lo, hi


def poisson_mean(poly):
    return sum((cp * cq * F(min(p, q), max(p, q))
                for p, cp in poly.items() for q, cq in poly.items()), F(0))
