from array import array
from fractions import Fraction as F
from functools import lru_cache

def mobius_sieve(limit):
    mu = array("b", [1]) * (limit + 1)
    mu[0] = 0
    composite = bytearray(limit + 1)
    primes = []
    for p in range(2, limit + 1):
        if composite[p]:
            continue
        primes.append(p)
        for n in range(p, limit + 1, p):
            composite[n] = 1
            mu[n] = -mu[n]
        if p * p <= limit:
            for n in range(p * p, limit + 1, p * p):
                mu[n] = 0
    prefix = array("i", [0]) * (limit + 1)
    for n in range(1, limit + 1):
        prefix[n] = prefix[n - 1] + mu[n]
    return mu, prefix, primes


def divisor_inverse(limit):
    out = [0] * (limit + 1)
    out[1] = 1
    for d in range(1, limit + 1):
        for n in range(2 * d, limit + 1, d):
            out[n] -= out[d]
    return out


def clean(poly):
    return {p: v for p, v in poly.items() if v}


def amplitude_coefficients(root, prefix, primes):
    endpoint = root * root - 1
    out = {1: F(prefix[root - 1] - prefix[endpoint])}
    for p in primes:
        if p * p >= root:
            break
        if p != 2:
            out[p] = -F(prefix[endpoint // (p * p)], p)
    return clean(out)


def step_coefficients(root, mu, prefix, primes, include_birth=True):
    lower = root * root - 1
    upper = (root + 1) ** 2 - 1
    shell = prefix[upper] - prefix[lower]
    c = mu[root] - shell
    daughters = {}
    for p in primes:
        if p * p > root:
            break
        if p == 2:
            continue
        if p * p < root:
            b = prefix[upper // (p * p)] - prefix[lower // (p * p)]
        else:
            b = prefix[root + 2] if include_birth else 0
        daughters[p] = F(b, p)
    return c, daughters


@lru_cache(None)
def log_interval(ratio):
    assert 1 <= ratio <= 2
    t = (ratio - 1) / (ratio + 1)
    terms = 32
    lower = 2 * sum((t ** (2 * k + 1) / (2 * k + 1)
                     for k in range(terms)), F(0))
    error = 2 * t ** (2 * terms + 1) / (
        (2 * terms + 1) * (1 - t * t))
    return lower, lower + error


def triangular_kernel_interval(p, q):
    ratio = F(max(p, q), min(p, q))
    if ratio >= 2:
        return F(0), F(0)
    if ratio == 1:
        return F(1), F(1)
    lo, hi = log_interval(ratio)
    log2lo, log2hi = log_interval(F(2))
    return 1 - hi / log2lo, 1 - lo / log2hi

    for p, cp in poly.items():
        for q, cq in poly.items():
            kl, kh = kernel(p, q)
            factor = cp * cq
            if factor >= 0:
                lo += factor * kl
                hi += factor * kh
            else:
                lo += factor * kh
                hi += factor * kl
    return lo, hi

                for p, cp in poly.items() for q, cq in poly.items()), F(0))

from fractions import Fraction as F

from functools import lru_cache


def clean(x):
    return {k: v for k, v in x.items() if v}


def add(*xs):
    out = {}
    for x in xs:
        for k, v in x.items():
            out[k] = out.get(k, F(0)) + v
    return clean(out)


def scale(x, c):
    return clean({k: v * c for k, v in x.items()})


ONE = {0: F(1)}


@lru_cache(None)
def factors(n):
    out = {}
    d = 2
    while d * d <= n:
        while n % d == 0:
            out[d] = out.get(d, 0) + 1
            n //= d
        d += 1
    if n > 1:
        out[n] = out.get(n, 0) + 1
    return out


def log(n):
    return {p: F(e) for p, e in factors(n).items()}


def mu(n):
    ff = factors(n)
    return 0 if any(e > 1 for e in ff.values()) else (-1) ** len(ff)


def mangoldt(n):
    ff = factors(n)
    return {next(iter(ff)): F(1)} if len(ff) == 1 else {}


def kernel(B, k):
    return scale(add(log(B), scale(log(k), -1)), mu(k))


def V(B):
    return add(ONE, *(scale(kernel(B, k), -F(1, k))
                      for k in range(1, B + 1)))


def tail(B, m):
    return add(*(scale(kernel(B, k), m // k) for k in range(B + 1, m + 1)))


def square(x):
    out = {}
    for k, v in x.items():
        for j, w in x.items():
            key = tuple(sorted((k, j)))
            out[key] = out.get(key, F(0)) + v * w
    return clean(out)

    return add(*(scale(square(values[m]), F(1, m * (m + 1)))
                 for m in range(lo, hi + 1)))
