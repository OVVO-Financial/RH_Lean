def mobius_table(n):
    mu = [1] * (n + 1)
    mu[0] = 0
    primes = []
    composite = bytearray(n + 1)

        if composite[p]:
            continue
        primes.append(p)
        for k in range(p, n + 1, p):
            mu[k] = -mu[k]
            if k > p:
                composite[k] = 1
        for k in range(p * p, n + 1, p * p):
            mu[k] = 0

def kappa(y, a, p, r):
    return int(a <= y) - int(p*a <= y) - int(r*a <= y) + int(p*r*a <= y)


def cap_options(g, q, anchor, count):
    parity = (anchor + count) % 2
    k = (g - 2*q*parity) // (4*q)
    lo, hi = anchor-count, anchor+count
    options = sorted({max(lo, min(hi, parity+2*j)) for j in (k, k+1)})
    score = lambda d: F(2*g*d, q) - 2*d*d
    best = max(map(score, options))
    return [d for d in options if score(d) == best]

    result = {(1, 0, 0), (0, 1, 0), (0, 0, 1)}

def sieve(limit):
    least = [0] * (limit + 1)
    mu = [0] * (limit + 1)
    mu[1] = 1
    primes = []

            least[n] = n
            primes.append(n)
            mu[n] = -1
        for p in primes:
            if p * n > limit:
                break
            least[p * n] = p
            if n % p == 0:

            mu[p * n] = -mu[n]

def factor(n):
    out = {}
    p = 2
    while p * p <= n:
        while n % p == 0:
            out[p] = out.get(p, 0) + 1
            n //= p
        p += 1
    if n > 1:
        out[n] = out.get(n, 0) + 1
    return out

    h = [0] * (x + 1)

        h = [v + w for v, w in zip(a, dh)]
    return h

                    out.append(((j, c, factors), product, sign))
                return

            else:

    return out
