def mobius_table(n):
    mu = [1] * (n + 1)
    mu[0] = 0
    primes = []
    composite = bytearray(n + 1)

        if composite[p]:
            continue
        primes.append(p)
        for k in range(p, n + 1, p):
            mu[k] = -mu[k]
            if k > p:
                composite[k] = 1
        for k in range(p * p, n + 1, p * p):
            mu[k] = 0

def kappa(y, a, p, r):
    return int(a <= y) - int(p*a <= y) - int(r*a <= y) + int(p*r*a <= y)


def cap_options(g, q, anchor, count):
    parity = (anchor + count) % 2
    k = (g - 2*q*parity) // (4*q)
    lo, hi = anchor-count, anchor+count
    options = sorted({max(lo, min(hi, parity+2*j)) for j in (k, k+1)})
    score = lambda d: F(2*g*d, q) - 2*d*d
    best = max(map(score, options))
    return [d for d in options if score(d) == best]

    result = {(1, 0, 0), (0, 1, 0), (0, 0, 1)}
