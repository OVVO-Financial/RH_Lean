def factor(n):
    out = []
    p = 2
    while p * p <= n:
        if n % p == 0:
            e = 0
            while n % p == 0:
                n //= p
                e += 1
            out.append((p, e))
        p += 1
    if n > 1:
        out.append((n, 1))
    return out


def trial_tables(x):
    mu, primes = [0] * (x + 1), []
    mu[1] = 1
    for n in range(2, x + 1):
        fs = factor(n)
        mu[n] = 0 if any(e > 1 for _, e in fs) else (-1) ** len(fs)
        if fs == [(n, 1)]:
            primes.append(n)
    return mu, primes


def column(a, cutoffs, divs):
