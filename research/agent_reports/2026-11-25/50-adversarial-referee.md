from fractions import Fraction as F
from itertools import combinations, product
from math import ceil, isqrt


def prime(p):
    return p >= 2 and all(p % d for d in range(2, isqrt(p) + 1))


def bit(a, p):
    assert a % p
    return int(pow(a, (p - 1) // 2, p) == p - 1)


def canon(a, mod):
    return min(a % mod, -a % mod)


def rank_mod(rows, prime_mod):
    pivots = {}
    for row in rows:
        v = [a % prime_mod for a in row]
        for j in range(len(v)):
            if not v[j]:
                continue
            if j not in pivots:
                inv = pow(v[j], -1, prime_mod)
                pivots[j] = [a * inv % prime_mod for a in v]
                break
            s = v[j]
            v = [(a - s * b) % prime_mod for a, b in zip(v, pivots[j])]
    return len(pivots)


def matrix(e, weights=(1, 1, 1), orbit_prime=5):
    n = 2**e
    mod = 4 * n
    classes = [canon(pow(orbit_prime, k, mod), mod) for k in range(n)]
    assert len(set(classes)) == n
    tops = set(classes[::2])
    rows = []
    for k in range(0, n, 2):
        for i, weight in zip((1, 3, 5), weights):
            target = canon(classes[k] * orbit_prime**i, mod)
            assert target == classes[(k + i) % n] and target not in tops
            row = [0] * n
            row[k] = -weight
            row[(k + i) % n] = 1
            rows.append(row)
    return rows


def det(a):
    m = [list(row) for row in a]
    sign, prev = 1, 1
    for k in range(len(m) - 1):
        i = next((j for j in range(k, len(m)) if m[j][k]), None)
        if i is None:
            return 0
        if i != k:
            m[k], m[i] = m[i], m[k]
            sign = -sign
        pivot = m[k][k]
        for i in range(k + 1, len(m)):
            for j in range(k + 1, len(m)):
                x = m[i][j] * pivot - m[i][k] * m[k][j]
                assert x % prev == 0
                m[i][j] = x // prev
        prev = pivot
    return sign * m[-1][-1]


def val5(a):
    if not a:
        return float("inf")
    r = 0
    while a % 5 == 0:
        r += 1
        a //= 5
    return r


def height_test(e, q, d, height, plus, xi=F(1, 4)):
    # Direct hinges on the CRT image. Maintain a rational coefficient per log p.
    n, mod = 2**e, 2 ** (e + 2)
    units = [canon(pow(5, k, mod), mod) for k in range(n)]
    idx = {v: i for i, v in enumerate(units)}
    support = set(range(n // 2)) if plus else set(range(n // 2, n))
    law = [F(1, n) - (F(1, len(support)) if u in support else 0) for u in range(n)]
    b = F(height + 1, height)
    a = [F(height - t % 2) + (xi if t == 0 else 0) for t in range(n)]
    N = ceil(max(a))
    co = {}
    for p in range(2, N + 1):
        if not prime(p):
            continue
        l = p
        while l <= N:
            db = [0, 1] if p == d else [bit(l, d)]
            qb = [0, 1] if p == q else [bit(l, q)]
            dh = range(n) if p == 2 else [idx[canon(l, mod)]]
            mass = F(2, len(db) * len(qb) * len(dh))
            for x, y, h in product(db, qb, dh):
                if x != 1 or y != 0:
                    continue  # Plane 0 has width 1 on every base; its law difference is zero.
                for t in range(n):
                    pos = (h + t) % n
                    hinge = max(0, a[t] / l - b) - max(0, a[t] / l - 1)
                    co[p] = co.get(p, 0) + mass * law[pos] * hinge
            l *= p
    # The sum of log-widths is the same on every base; minus MAIN cancels.
    return {p: F(2, 1) * v / b for p, v in co.items() if v}


assert prime(5) and prime(3) and prime(23) and bit(5, 23) == bit(5, 3) == 1
assert pow(-23, 1, 3) == 1 and 3 % 8 == 3 and 23 % 16 == 7
mass = sum(max(0, F(3126, 5**i) - 1) for i in (1, 3, 5, 7))
assert mass == F(2025651, 3125)
print("Positive normalized 5-component:", 4 * mass, "* log(5)")
for e in range(3, 9):
    a = matrix(e)
    assert rank_mod(a, 5) == rank_mod(a, 101) == 2**e - 1
    print("e, rows, incidence rank, row deficiency:", e, len(a), 2**e - 1, len(a) - 2**e + 1)
assert prime(53) and bit(53, 3) == bit(53, 23) == 1
for e in range(3, 9):
    a = matrix(e, orbit_prime=53)
    assert rank_mod(a, 53) == 2**e - 1
print("p=53: distinct retained targets and rank verified at e=3,...,8")

tests = minors = 0
for A, B, C in [(1, 1, 1), (1, 1, 6), (1, 1, 26), (1, 1, 126),
                 (1, 2, 4), (2, 1, 3), (2, 3, 2), (3, 4, 1), (1, 2, 3)]:
    a = matrix(3, (A, B, C))
    expected = min(val5(B**4 - A**4), val5(A * C - B**2))
    vals = [val5(det(rows)) for rows in combinations(a, 8)]
    assert min(vals) == expected
    assert rank_mod(a, 5) == (7 if expected > 0 else 8)
    tests += 1
    minors += len(vals)
    print("A,B,C; predicted/observed last valuation:", A, B, C, expected, min(vals))
print("profiles, maximal minors:", tests, minors)

replayed = [(2, 11, 7, 49, True), (2, 11, 7, 49, False), (2, 11, 7, 50, True),
            (3, 3, 23, 529, True)]
expected = [{7: F(1, 400)}, {7: -F(1, 400)}, {}, {23: -F(1, 8480)}]
for t, want in zip(replayed, expected):
    got = height_test(*t)
    assert got == want, (got, want)
    print("HEIGHTFIBRE4, normalized coefficients:", t, got)
