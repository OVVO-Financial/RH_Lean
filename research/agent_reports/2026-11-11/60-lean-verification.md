"""Registered implementation checks: no Lean or infinite-scale inference.

Compare the new labelled recurrence with a direct finite geometric polynomial,
then average and multiply ALL terms. Fixed domain: base 3; log-delay exponents
8 and 16; A in {-1, 0, 1}; eta in {-1, 1}; K=0..3; theta=5/8,3/4,7/8.
Pool weights 1/3,2/3. Test the empty product and a nonzero mixed-term control.
"""

from fractions import Fraction as F
from itertools import product
from hashlib import sha256
from pathlib import Path


def add(p, q):
    return p + q


def scale(c, p):
    return [(n, c * a) for n, a in p]


def mul(p, q):
    return [(n + m, a * b) for n, a in p for m, b in q]


def collect(p):
    d = {}
    for n, a in p:
        d[n] = d.get(n, F(0)) + a
    return {n: a for n, a in d.items() if a}


def mass(p, theta):
    total = F(0)
    for n, a in p:
        e = -n * theta
        assert e.denominator == 1
        total += abs(a) * F(3) ** int(e)
    return total


def local(e, A, eta, K):
    Q = F(3) ** e
    q = 1 / Q
    H = [(6 * e, A * Q**2), (e, (1 - q) * eta)]
    W = [(5 * e, A * eta * Q**3)]
    R0 = scale(-(1 - q), W)
    R = R0
    for _ in range(K):
        R = add(R0, mul(H, add([(0, 1 - q)], R)))
    G, term = [(0, F(1))], [(0, F(1))]
    for _ in range(K):
        term = mul(term, H)
        G = add(G, term)
    direct = add([(0, q)], scale(1 - q, mul(add([(0, F(1))], scale(-1, W)), G)))
    assert collect(add([(0, F(1))], R)) == collect(direct)
    assert all(n > 0 for n, _ in R)
    return R


def complete(ps):
    R = []
    for p in reversed(ps):
        R = add(add(p, R), mul(p, R))
    return R


checks = 0
for e, A, eta, K in product((8, 16), (-1, 0, 1), (-1, 1), range(4)):
    R = local(e, F(A), F(eta), K)
    for theta in (F(5, 8), F(3, 4), F(7, 8)):
        r = F(3) ** int(e * (2 - 6 * theta)) + F(3) ** int(-e * theta)
        w = F(3) ** int(e * (3 - 5 * theta))
        assert r <= F(1, 2)
        assert mass(R, theta) <= 2 * (r + w)
        checks += 1
mixed_checks = 0
for K, theta in product(range(3), (F(5, 8), F(3, 4), F(7, 8))):
    ps = [add(scale(F(1, 3), local(e, F(1), F(1), K)),
              scale(F(2, 3), local(e, F(-1), F(-1), K))) for e in (8, 16)]
    R = complete(ps)
    assert collect([(0, F(1))] + R) == collect(mul([(0, F(1))] + ps[0], [(0, F(1))] + ps[1]))
    assert 1 + mass(R, theta) == (1 + mass(ps[0], theta)) * (1 + mass(ps[1], theta))
    assert collect(R) != collect(ps[0] + ps[1])
    mixed_checks += 1
assert complete([]) == []
