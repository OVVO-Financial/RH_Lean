from itertools import product
from hashlib import sha256

def add(p, q):
    return p + q


def scale(c, p):
    return [(n, c * a) for n, a in p]


def mul(p, q):
    return [(n + m, a * b) for n, a in p for m, b in q]


def collect(p):
    d = {}
    for n, a in p:
        d[n] = d.get(n, F(0)) + a
    return {n: a for n, a in d.items() if a}


def mass(p, theta):
    total = F(0)
    for n, a in p:
        e = -n * theta
        assert e.denominator == 1
        total += abs(a) * F(3) ** int(e)
    return total


def local(e, A, eta, K):
    Q = F(3) ** e
    q = 1 / Q
    H = [(6 * e, A * Q**2), (e, (1 - q) * eta)]
    W = [(5 * e, A * eta * Q**3)]

        term = mul(term, H)
        G = add(G, term)
    direct = add([(0, q)], scale(1 - q, mul(add([(0, F(1))], scale(-1, W)), G)))
    assert collect(add([(0, F(1))], R)) == collect(direct)
    assert all(n > 0 for n, _ in R)
    return R

    R = []
    for p in reversed(ps):
        R = add(add(p, R), mul(p, R))
