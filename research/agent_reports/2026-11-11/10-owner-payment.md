    lo, hi = log_unit(a, b)
    return lo+k*LOG2[0], hi+k*LOG2[1]


def ei_bounds(tl, th):
    lo, hi = log_bounds(tl, S)[0], log_bounds(th, S)[1]
    pl = ph = S
    terms = 180
    for k in range(1, terms+1):
        pl, ph = pl*tl//(S*k), ceildiv(ph*th, S*k)
        lo += pl//k
        hi += ceildiv(ph, k)
    nxt = ceildiv(ph*th, S*(terms+1))
    assert th < S*(terms+2)
    hi += ceildiv(nxt*S*(terms+2), (terms+1)*(S*(terms+2)-th))
    return lo, hi


EI2 = ei_bounds(*LOG2)


@lru_cache(None)
def li_bounds(n):
    if n == 2:
        return 0, 0
    lo, hi = ei_bounds(*log_bounds(n))
    return lo-EI2[1], hi-EI2[0]


def potential(n):
    lo, hi = li_bounds(n)
    assert lo//S == hi//S
    return lo//S
```

                exponent += 1
            primes.append(p)

        p += 1
