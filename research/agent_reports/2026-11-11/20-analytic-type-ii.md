def optimizer(n):
    s = sum((F(PHI[d] ** 2, J2[d]) for d in range(2, n + 1)), F(0))
    q = [F(0), F(0)] + [F(2 * PHI[d], J2[d]) / s for d in range(2, n + 1)]
    c = invert(q)
    assert coordinates(c)[2:] == q[2:]
    assert sum((c[k] * F(k - 1, 2 * k) for k in range(2, n + 1)), F(0)) == 1
    assert variance(q) == 1 / (3 * s)
    assert s <= n - 1
    if n >= 20:
        assert s >= F(n, 20)
    if n >= 3:
        for t in (F(-2), F(1, 3), F(5)):
            qp = q.copy()
            qp[2] += 2 * t
            qp[3] -= t
            assert sum((PHI[d] * qp[d] for d in range(2, n + 1)), F(0)) == 2
            assert variance(qp) == variance(q) + sum((J2[d] * (qp[d] - q[d]) ** 2 for d in range(2, n + 1)), F(0)) / 12
            COUNTS['optimizer_square_completion'] += 1
    b = sum(abs(x) for x in c)
    assert b <= 4 * n / s
    for k in range(2, n + 1):
        assert abs(c[k]) <= 2 * harmonic(n // k) / s
        COUNTS['optimizer_coefficient_bounds'] += 1
    if n <= 8:
        period = lcm(*range(2, n + 1))
        average = sum((residual(c, m) ** 2 for m in range(period)), F(0)) / period
        assert average == variance(q)
        COUNTS['optimizer_full_periods'] += 1
    return c, s, variance(q)


def full_norm_interval(c, tmax=100000):
    n = len(c) - 1
    q1 = sum((c[k] / k for k in range(2, n + 1)), F(0))
    denominator = lcm(q1.denominator, *(x.denominator for x in c))
    drift = int(q1 * denominator)
    jumps = [0] * (tmax + 1)
    for k in range(2, n + 1):
        ck = int(c[k] * denominator)
        for m in range(k, tmax + 1, k):
            jumps[m] += ck
    scale = 1 << 80
    rnum, lower, upper = denominator, 0, 0
    for m in range(1, tmax + 1):
        rnum += jumps[m] - drift
        if m <= 3 * n:
            assert F(rnum, denominator) == residual(c, m)
            COUNTS['norm_recurrence_checks'] += 1
        num = rnum * rnum * scale
        den = denominator * denominator * m * (m + 1)
        quotient, rem = divmod(num, den)
        lower += quotient
        upper += quotient + bool(rem)
    b = sum(abs(x) for x in c)
    tail = (1 + b) ** 2 / (tmax + 1)
    COUNTS['outward_rational_norm_terms'] += tmax
    return F(lower, scale), F(upper, scale) + tail
