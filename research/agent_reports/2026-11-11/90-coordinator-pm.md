"""Exact checks for 20261111-coordinator-0710-EST/FL1-FL4.

Finite validation only. The infinite Gram and full-error implications have
separate paper proofs; no finite table is used to infer an infinite estimate.
"""

from collections import Counter, defaultdict
from fractions import Fraction as F
from math import comb, factorial, gcd, isqrt
import json


def factors(n):
    out = {}
    p = 2
    while p * p <= n:
        while n % p == 0:
            out[p] = out.get(p, 0) + 1
            n //= p
        p += 1
    if n > 1:
        out[n] = out.get(n, 0) + 1
    return out


def mu(n):
    fs = factors(n)
    return 0 if any(e > 1 for e in fs.values()) else (-1) ** len(fs)


def divisors(n):
    return [d for d in range(1, n + 1) if n % d == 0]


def jordan(n):
    out = n * n
    for p in factors(n):
        out = out // (p * p) * (p * p - 1)
    return out


def sigma(n):
    return sum(divisors(n))


def layer(k, j):
    return k >= 2 and mu(k) != 0 and len(factors(k)) == j


def bj(j):
    return sum((F(comb(j, s), 2**s * factorial(s)) for s in range(j + 1)), F(0))


def gram_formula(k, l):
    g = gcd(k, l)
    return F(mu(k) * mu(l) * sum(d*d for d in divisors(g)), k*l)


def mu_power(e):
    return 1 if e == 0 else -1 if e == 1 else 0


def gram_euler(k, l):
    fk, fl = factors(k), factors(l)
    out = F(k*l)
    for p in set(fk) | set(fl):
        a, b = fk.get(p, 0), fl.get(p, 0)
        local = F(0)
        for e in range(max(a, b), min(a, b) + 2):
            jp = 1 if e == 0 else p ** (2*e-2) * (p*p-1)
            local += F(mu_power(e-a) * mu_power(e-b), jp)
        out *= (1-F(1, p*p)) * local
    return out


def psd_exact(matrix):
    a = [row[:] for row in matrix]
    for i in range(len(a)):
        pivot = a[i][i]
        assert pivot >= 0, (i, pivot)
        if pivot == 0:
            assert all(a[i][j] == 0 for j in range(i+1, len(a)))
            continue
        for j in range(i+1, len(a)):
            for k in range(j, len(a)):
                a[j][k] -= a[i][j] * a[i][k] / pivot
                a[k][j] = a[j][k]


def original_values(c, upto):
    return [sum((v * F(m % k, k) for k, v in c.items()), F(0))
            for m in range(upto+1)]


def variance(c):
    n = max(c, default=1)
    q = {d: sum((v/F(k) for k, v in c.items() if k % d == 0), F(0))
         for d in range(1, n+1)}
    vq = sum((jordan(d)*q[d]**2 for d in range(2, n+1)), F(0))/12
    vg = sum((ck*cl*F(gcd(k,l)**2-1, k*l)
              for k, ck in c.items() for l, cl in c.items()), F(0))/12
    assert vq == vg
    return vq, q


def check_triangle(qcount, cv, es):
    # sqrt(qcount) <= sqrt(cv) + 2 sqrt(es), without rounded roots.
    rem = F(qcount)-cv-4*es
    if rem > 0:
        assert rem*rem <= 16*cv*es


counts = Counter()
sf = [k for k in range(2, 161) if mu(k)]
for k in sf:
    for l in sf:
        assert gram_euler(k,l) == gram_formula(k,l)
        counts['gram_euler_entries'] += 1

small = [k for k in range(2,25) if mu(k)]
zeta_upper = sum((F(1,n*n) for n in range(1,65)),F(0))+F(1,64)
mat = []
for k in small:
    row = []
    for l in small:
        direct = sum((F(k*l*mu(d//k)*mu(d//l), jordan(d))
                      for d in range(2,129) if d % k == 0 and d % l == 0),F(0))
        row.append(zeta_upper*gram_formula(k,l)-direct)
    mat.append(row)
psd_exact(mat)
counts['truncated_gram_psd'] += 1

for n in range(2,65):
    vectors = [
        {k:F(0) for k in range(2,n+1)},
        {k:F((k*7+3)%11-5, k%4+1) for k in range(2,n+1)},
        {k:F(mu(k)) for k in range(2,n+1)},
        {k:F((-1)**k, k) for k in range(2,n+1)},
        {k:F(k if k == n else 0) for k in range(2,n+1)},
    ]
    for c in vectors:
        v,q = variance(c)
        counts['covariance_vectors'] += 1
        vals = original_values(c, 2*n)
        residual = [1-u for u in vals]
        energies = [F(0)]
        for m in range(1,2*n+1):
            energies.append(energies[-1]+residual[m]**2/F(m*(m+1)))
        for k in range(2,2*n+1):
            ck = c.get(k,F(0))
            inv = k*sum((mu(t)*q.get(k*t,F(0))
                         for t in range(1,n//k+1)), F(0))
            assert inv == ck
            recovered = -mu(k)+mu(k)*residual[1]+sum(
                (mu(k//d)*(residual[d]-residual[d-1])
                 for d in divisors(k) if d>=2), F(0))
            assert recovered == ck
            assert (ck+mu(k))**2 <= 4*sigma(k)**2*energies[k]
            counts['coefficient_recoveries'] += 1
        for j in range(1,5):
            total = sum((ck*ck for k,ck in c.items() if layer(k,j)),F(0))
            assert total <= 20*bj(j)*v
            counts['layer_bessel_bounds'] += 1
            h=2*n
            inds=[k for k in range(2,h+1) if layer(k,j)]
            sig2=sum(sigma(k)**2 for k in inds)
            check_triangle(len(inds),20*bj(j)*v,energies[h]*sig2)
            counts['joint_head_error_bounds'] += 1

for h in [32,64,128,256,512]:
    for j in range(1,5):
        inds=[k for k in range(2,h+1) if layer(k,j)]
        for t in range(4):
            z={k:F(((k+3)*(t+2))%13-6, (k+t)%3+1) for k in inds}
            features=defaultdict(F)
            for k in inds:
                for d in divisors(k):
                    features[d]+=F(d,k)*z[k]
            quad=sum((v*v for v in features.values()),F(0))
            norm=sum((v*v for v in z.values()),F(0))
            assert quad <= bj(j)*norm
            if h<=64:

                            for k in inds for l in inds), F(0))
                assert direct==quad
                counts['gram_feature_comparisons']+=1
            counts['homogeneous_gram_bounds']+=1

head_rows=[]
for h in [2,3,5,8,16,32,64,128]:
    n=h+1
    c={k:F(-mu(k)) for k in range(2,h+1)}
    c[n]=n*(1+sum((F(mu(k),k) for k in range(2,h+1)),F(0)))

    assert all(vals[m]==1 for m in range(1,h+1))

    assert q[1]==1
    for j in range(1,5):
        qj=sum(layer(k,j) for k in range(2,h+1))
        assert qj<=20*bj(j)*v
        counts['exact_head_layer_bounds']+=1
    head_rows.append({'H':h,'Q2':sum(layer(k,2) for k in range(2,h+1)),
                      'variance':str(v),'semiprime_floor':str(F(2,85)*sum(layer(k,2) for k in range(2,h+1)))})
    counts['exact_heads']+=1

assert gram_formula(6,10)==F(1,12) != F(1,60)
assert gram_formula(2,6)==-F(5,12)<0
assert gram_euler(2,2)==F(5,4) != 1
counts['negative_controls']=3
assert bj(1)==F(3,2) and bj(2)==F(17,8)
assert 20*bj(2)==F(85,2)
assert sum((F(1,n*n) for n in range(1,6)),F(0))+F(1,5)<F(5,3)
counts['rational_constants']=4

print(json.dumps({'status':'PASS','counts':dict(counts),'head_rows':head_rows,
                  'constant_j2':str(20*bj(2)),
                  'gram_controls':{'6,10':str(gram_formula(6,10)),
                                   '2,6':str(gram_formula(2,6)),
                                   '2,2':str(gram_formula(2,2))}},indent=2))

from io import StringIO

    import nyman_factor_layers as base


def gcd_kernel(k, l):
    return F(gcd(k, l)**2, k*l)


def row_bound(h):
    return max(sum((gcd_kernel(k, l) for l in range(1, h+1)), F(0))
               for k in range(1, h+1))


counts = {}
sf = [k for k in range(2, 161) if base.mu(k)]

        assert abs(base.gram_formula(k, l)) <= F(5, 3)*gcd_kernel(k, l)
