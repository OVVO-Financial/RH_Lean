        assert (p, q) in rectangle(r) and (q, s) in rectangle(2 * r)
        x, y = r * r - 1, 4 * r * r - 1
        assert p * q <= x and q * s <= y
        ratio = F(x * s, y * p)
        epsilon = abs(ratio - 1)
        bound = epsilon / (1 - epsilon)
        lower = 1 - 2 * r * bound * bound
        assert lower > F(1, 2)
        witnesses.append(dict(R=r, ratio=str(ratio), cosine_lower=str(lower)))
    match_rows = []
    for r in [16, 20, 30, 64, 128]:
        x = r * r - 1
        h = math.isqrt(r)
        h += h * h < r
        dmax = x // (4 * h)
        products = {p * q for p, q in rectangle(r)}
        ds = [d for d in range(1, dmax + 1)
              if x - d in products and x + d in products]
        upper = sum((F(h * d * d, 3 * (x * x - d * d)) for d in ds), F(0))
        assert upper <= F(4 * r, 765)
        match_rows.append(dict(R=r, D=dmax, matched=len(ds), rational_upper=str(upper)))
    root_rows = []
    for r in range(2, 65):
        small = [p for p in range(2, r) if prime(p)]
        phi = lambda n: sum(all(k % p for p in small) for k in range(1, n + 1))
        got = phi(r + 2) - 1 if prime(r) else 0
        want = 3 if r == 2 else int(prime(r)) * (1 + int(prime(r + 2)))
        assert got == want
        root_rows.append(got)
    return dict(phase_roots=len(roots), prime_products=pair_count,
                resonance_roots=resonances, shared_prime_witnesses=witnesses,
                reflected_matches=match_rows, root_admission_roots=len(root_rows),
                root_admission_sum=sum(root_rows))


def log_enclosure(x, n=60):
    z = (x - 1) / (x + 1)
    mid = 2 * sum((z ** (2 * k + 1) / (2 * k + 1) for k in range(n)), F(0))
    tail = 2 * abs(z) ** (2 * n + 1) / ((2 * n + 1) * (1 - z * z))
    return mid, tail


def pi_kernel_16(product):
    # For R=16 the full band is on the first polynomial piece of b.
    phase, radius = log_enclosure(F(255, product))
    value = F(0)
    nterms = 32
    for k in range(nterms):
        power = 2 * k
        moment = F(8 ** (power + 1) - 4 ** (power + 1), power + 1)
        moment -= F(6, 16 ** 2) * F(8 ** (power + 3) - 4 ** (power + 3), power + 3)
        moment += F(6, 16 ** 3) * F(8 ** (power + 4) - 4 ** (power + 4), power + 4)
        value += (-1) ** k * phase ** (2 * k + 1) * moment / math.factorial(2 * k + 1)
    remainder = (8 * abs(phase)) ** (2 * nterms + 1) / math.factorial(2 * nterms + 1)
    return value, remainder + 4 * radius


def rational_sign_witness():
    rows = [(p * q, pi_kernel_16(p * q)) for p, q in rectangle(16)]
    out = {}
    for name, selected in [("all", rows), ("collar", [v for v in rows if abs(v[0] - 255) <= 15])]:
        value = sum((v for _, (v, _) in selected), F(0))
        error = sum((e for _, (_, e) in selected), F(0))
        lo, hi = value - error, value + error
        assert hi < 0 if name == "all" else lo > 0
        scale = 10 ** 12
        out[name] = dict(products=[p for p, _ in selected],
                         pi_J_lower_scaled=(lo * scale).__floor__(),
                         pi_J_upper_scaled=(hi * scale).__ceil__(), scale=scale)
    return out


def packet_parameters(r):
    ell = math.log(2)
    u = math.sqrt(r)
    n = math.ceil(5 * u * ell / (8 * math.pi))
    return r / (4 * math.log(r)), 2 * math.pi * n / ell


def sigma_prefix(x, start=4096):
    x = np.asarray(x, dtype=float)
    result = np.zeros_like(x)
    r = start
    while r / 2 <= x.max():
        c, t0 = packet_parameters(r)
        left = (r / 2 <= x) & (x < r)
        right = (r <= x) & (x < 2 * r)
        result[left] = -c * np.sin(t0 * np.log(r / x[left])) / t0
        result[right] = c * (1 - np.cos(t0 * np.log(x[right] / r))) / t0
        r *= 4
    return result


def interval_nodes(lo, hi, order, width=2):
    z, w = leggauss(order)
    edges = np.linspace(lo, hi, max(1, math.ceil((hi - lo) / width)) + 1)
    xs, ws = [], []
    for a, b in zip(edges[:-1], edges[1:]):
        xs.extend((a + b) / 2 + (b - a) / 2 * z)
        ws.extend((b - a) / 2 * w)
    return np.array(xs), np.array(ws)


def reference_transform(t, r, left):
    ell, lr = math.log(2), math.log(r)
    lo, hi = (-ell, 0) if left else (0, ell)
    s, w = interval_nodes(lo, hi, 24, width=0.025)
    return np.exp(-1j * t[:, None] * s) @ (w * r * np.exp(s) / (lr + s))


def atom_transform(t, r, atoms, left):
    selected = atoms[(atoms >= (r / 2 if left else r)) &
                     (atoms < (r if left else 2 * r))]
    values = np.zeros(len(t), dtype=complex)
    for offset in range(0, len(selected), 512):
        logs = np.log(selected[offset:offset + 512] / r)
        values += np.exp(-1j * t[:, None] * logs).sum(axis=1)
    return values


def model_diagnostics():
    # These floors and transforms are floating diagnostics, not certified atoms.
    sites = np.arange(2049, 131072, 2, dtype=np.int64)
    lam = expi(np.log(sites.astype(float))) - expi(math.log(2048)) + sigma_prefix(sites)
    levels = np.floor(lam).astype(np.int64)
    increments = np.diff(np.concatenate(([0], levels)))
    assert np.isin(increments, [0, 1]).all()
    atoms = sites[increments == 1].astype(float)
    nearest = float(np.min(np.minimum(lam - levels, 1 - (lam - levels))))
    rows = []
    ell = math.log(2)
    for r in [4096, 16384, 65536]:
        c, t0 = packet_parameters(r)
        results = []
        for order in [16, 24]:
            u = math.sqrt(r)
            t, wt = interval_nodes(u, 2 * u, order)
            fourier = lambda v: ell * np.exp(-1j * v * ell / 2) * np.sinc(v * ell / (2 * np.pi))
            g, h = fourier(t - t0), fourier(t + t0)
            sa = c * (np.conj(g) + np.conj(h)) / 2
            sb = c * (g - h) / (2j)
            aa = atom_transform(t, r, atoms, True) - reference_transform(t, r, True)
            ab = atom_transform(t, r, atoms, False) - reference_transform(t, r, False)
            spline = 1 - 6 * (t / r) ** 2 + 6 * (t / r) ** 3
            phase = np.exp(1j * t * math.log1p(-1 / r ** 2))
            band = lambda a, b: float(np.sum(wt * spline * np.imag(phase * a * b) / (math.pi * t)))
            ms, ma = band(sa, sb), band(aa, ab)
            norms = [float(np.sqrt(np.sum(wt * abs(aa - sa) ** 2))),
                     float(np.sqrt(np.sum(wt * abs(ab - sb) ** 2)))]
            assert max(norms) <= 16 * u
            assert abs(ma - ms) <= 8 * r / math.log(r) + 128 * u
            assert ms <= -r ** 1.5 / (32768 * math.log(r) ** 2)
            results.append(dict(M_sigma=ms, M_atoms=ma, rounding_L2=norms))
        rows.append(dict(R=r, t0=t0, coarse=results[0], refined=results[1],
                         refinement_difference=abs(results[0]["M_atoms"] - results[1]["M_atoms"])))
    return dict(status="FLOATING_DIAGNOSTIC_ONLY", min_floor_distance=nearest,
                atoms=int(len(atoms)), sites=int(len(sites)), rows=rows)


if __name__ == "__main__":
    print(json.dumps(dict(exact=exact_checks(), rational_R16=rational_sign_witness(),
                         model=model_diagnostics()), indent=2))
