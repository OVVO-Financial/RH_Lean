"""Proof-led actual-prime diagnostic; registered here before first execution.

Fixed roots: 8, 9, 11, 12, 37, 256. Each uses U=1, R/8, R/(2 log(R)^2).
Check both discrepancy transforms against the explicit mean-square bound,
then the nonconjugated signed band with weight 1 and the SC spline weight.
Gauss orders 96 and 144 must agree within 1e-8. This is not an interval proof.
"""

import json
import math

import numpy as np
from numpy.polynomial.legendre import leggauss


ROOTS = (8, 9, 11, 12, 37, 256)


def primes_through(n):
    return [p for p in range(2, n + 1)
            if all(p % d for d in range(2, math.isqrt(p) + 1))]


def grid(a, b, order):
    nodes, weights = leggauss(order)
    return a + (b - a) * (nodes + 1) / 2, weights * (b - a) / 2


def transform(root, primes, lo, hi, t, order):
    p = np.array(primes, dtype=float)
    discrete = np.exp(-1j * t[:, None] * np.log(p[None, :] / root)).sum(axis=1)
    s, ws = grid(math.log(lo / root), math.log(hi / root), order)
    density = root * np.exp(s) / (math.log(root) + s)
    continuous = np.exp(-1j * t[:, None] * s[None, :]) @ (ws * density)
    return discrete - continuous


def evaluate(root, U, order):
    primes = primes_through(2 * root)
    left = [p for p in primes if root <= 2 * p and p < root]
    right = [p for p in primes if root <= p <= 2 * root]
    L = math.log(root)
    assert max(len(left), len(right)) <= 4 * root / L
    t, wt = grid(U, 2 * U, order)
    A = transform(root, left, root / 2, root, t, order)
    B = transform(root, right, root, 2 * root, t, order)

class Quadrature:
    def __init__(self, order):
        self.nodes, self.weights = leggauss(order)

    def integral(self, f, a, b):
        a, b = np.broadcast_arrays(np.asarray(a), np.asarray(b))
        h = np.maximum(b - a, 0) / 2
        x = (a + h)[..., None] + h[..., None] * self.nodes
        return h * np.sum(f(x) * self.weights, axis=-1)

    def li(self, a, b):
        return self.integral(lambda x: 1 / np.log(x), a, b)

    def li_grid(self, n):
        result = np.zeros(n + 1)
        starts = np.arange(2, n)
        result[3:] = np.cumsum(self.li(starts, starts + 1))
        return result

    def continuous_pair(self, r, cutoff):
        lo = r / 2
        hi = min(r, cutoff / r)
        if hi <= lo:
            return 0.0
        return float(self.integral(
            lambda x: self.li(r, np.minimum(2 * r, cutoff / x)) / np.log(x),
            lo, hi,
        ))

    def strip_pair(self, r):
        x = r * r - 1
        split = x / r
        total = self.integral(
            lambda u: self.li(x / u, r * r / u) / np.log(u), r / 2, split
        )
        total += self.integral(
            lambda u: self.li(r, r * r / u) / np.log(u), split, r
        )
        return float(total)


def actual(r, primes, quad, li_values):
    left = np.arange((r + 1) // 2, r)
    right = np.arange(r, 2 * r + 1)
    ps, qs = left[primes[left]], right[primes[right]]
    x = r * r - 1
    products = ps[:, None] * qs[None, :]
    assert not np.any(products == r * r)
    pp = int(np.count_nonzero(products <= x))
    equal_x = [(int(p), int(q)) for p in ps for q in qs if p * q == x]
    pl = float(np.sum(quad.li(r, np.maximum(r, np.minimum(2 * r, x / ps)))))
    lp = float(np.sum(quad.li(r / 2, np.maximum(r / 2, np.minimum(r, x / qs)))))
    ll = quad.continuous_pair(r, x)
    q = pp - pl - lp + ll

    sp = float(np.sum(quad.li(np.maximum(r, x / ps), np.minimum(2 * r, r * r / ps))))
    sl = float(np.sum(quad.li(np.maximum(r / 2, x / qs), np.minimum(r, r * r / qs))))
    strip = -sp - sl + quad.strip_pair(r)
    assert abs(strip) <= 16 / math.log(r) ** 2 + 1e-8

    floor_li = np.floor(li_values).astype(np.int64)

def smooth(r, order):
    ell, u, ll = math.log(2), math.sqrt(r), math.log(r)

        lambda s: c * c * np.cos(t * s) * (1 - np.cos(t * (s - delta))) / t,
        delta, ell,

                                  + math.sin(t * delta) / t)

def exact_controls():

class Poly:
    def __init__(self, terms=0):
        self.terms = terms if isinstance(terms, dict) else {(0, 0, 0, 0): F(terms)}
        self.terms = {k: v for k, v in self.terms.items() if v}

    def __add__(self, other):
        other = other if isinstance(other, Poly) else Poly(other)
        out = self.terms.copy()
        for k, v in other.terms.items():
            out[k] = out.get(k, F(0)) + v
        return Poly(out)

    __radd__ = __add__

    def __neg__(self):
        return Poly({k: -v for k, v in self.terms.items()})

    def __sub__(self, other):
        return self + (-other if isinstance(other, Poly) else -Poly(other))

    def __rsub__(self, other):
        return Poly(other) - self

    def __mul__(self, other):
        other = other if isinstance(other, Poly) else Poly(other)
        out = {}
        for k, v in self.terms.items():
            for l, w in other.terms.items():
                key = tuple(a + b for a, b in zip(k, l))
                out[key] = out.get(key, F(0)) + v * w
        return Poly(out)

    __rmul__ = __mul__

    def __truediv__(self, scalar):
        return self * (F(1) / scalar)
