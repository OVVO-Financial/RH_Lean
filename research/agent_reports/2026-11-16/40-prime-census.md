"""Independent product reconstruction and exact finite-state sign classification."""
import itertools
import json
import time
from collections import Counter
from fractions import Fraction
from functools import lru_cache
from math import gcd, isqrt
from pathlib import Path

start = time.perf_counter()
counts = Counter()

def check(name, condition):
    if not condition:
        raise AssertionError(name)
    counts[name] += 1

def factors(n):
    ans = []
    p = 2
    while p*p <= n:
        exponent = 0
        while n % p == 0:
            n //= p
            exponent += 1
        if exponent:
            ans.append((p, exponent))
        p += 1
    if n > 1:
        ans.append((n, 1))
    return ans

@lru_cache(None)
def trial_mu(n):
    fs = factors(n)
    return 0 if any(e > 1 for _, e in fs) else (-1)**len(fs)

def squarefree_divisors(fs):
    out = [(1, 1)]
    for p, _ in fs:
        out += [(p*d, -m) for d, m in out[:]]
    return out

def product_parameters(n):
    r, s, sign = 1, 1, 1
    for p, e in factors(n):
        r *= p
        if e > 1:
            s *= p
        else:
            sign = -sign
    return r, s, sign

limit = 1021**2 - 1
mu = [0]*(limit+1)
mu[1] = 1
for d in range(1, limit+1):
    if mu[d]:
        for n in range(2*d, limit+1, d):
            mu[n] -= mu[d]
M = [0]*(limit+1)
for n in range(1, limit+1):
    M[n] = M[n-1]+mu[n]

# Separate prime-divisor incidence construction of all product labels.
rad = [1]*(limit+1)
support = [1]*(limit+1)
single_sign = [1]*(limit+1)
for p in range(2, limit+1):
    if rad[p] != 1:
        continue
    for n in range(p, limit+1, p):
        rad[n] *= p
        single_sign[n] *= -1
    for n in range(p*p, limit+1, p*p):
        support[n] *= p
        single_sign[n] *= -1
for n in range(1, 4097):
    check('trial_mu', mu[n] == trial_mu(n))
    check('trial_labels', (rad[n], support[n], single_sign[n]) == product_parameters(n))

    fs = factors(n)

    radn, sn, sign = product_parameters(n)
    actual, intersection = Counter(), Counter()

def mobius(n):
    sign = 1
    p = 2
    while p*p <= n:
        if n%p==0:
            n//=p
            sign=-sign
            if n%p==0:
                return 0
        p+=1
    return -sign if n>1 else sign
