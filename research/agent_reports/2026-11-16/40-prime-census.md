"""Independent product reconstruction and exact finite-state sign classification."""
import itertools
import json
import time
from collections import Counter
from fractions import Fraction
from functools import lru_cache
from math import gcd, isqrt
from pathlib import Path

start = time.perf_counter()
counts = Counter()

def check(name, condition):
    if not condition:
        raise AssertionError(name)
    counts[name] += 1

def factors(n):
    ans = []
    p = 2
    while p*p <= n:
        exponent = 0
        while n % p == 0:
            n //= p
            exponent += 1
        if exponent:
            ans.append((p, exponent))
        p += 1
    if n > 1:
        ans.append((n, 1))
    return ans

@lru_cache(None)
def trial_mu(n):
    fs = factors(n)
    return 0 if any(e > 1 for _, e in fs) else (-1)**len(fs)

def squarefree_divisors(fs):
    out = [(1, 1)]
    for p, _ in fs:
        out += [(p*d, -m) for d, m in out[:]]
    return out

def product_parameters(n):
    r, s, sign = 1, 1, 1
    for p, e in factors(n):
        r *= p
        if e > 1:
            s *= p
        else:
            sign = -sign
    return r, s, sign

limit = 1021**2 - 1
mu = [0]*(limit+1)
mu[1] = 1
for d in range(1, limit+1):
    if mu[d]:
        for n in range(2*d, limit+1, d):
            mu[n] -= mu[d]
M = [0]*(limit+1)
for n in range(1, limit+1):
    M[n] = M[n-1]+mu[n]

# Separate prime-divisor incidence construction of all product labels.
rad = [1]*(limit+1)
support = [1]*(limit+1)
single_sign = [1]*(limit+1)
for p in range(2, limit+1):
    if rad[p] != 1:
        continue
    for n in range(p, limit+1, p):
        rad[n] *= p
        single_sign[n] *= -1
    for n in range(p*p, limit+1, p*p):
        support[n] *= p
        single_sign[n] *= -1
for n in range(1, 4097):
    check('trial_mu', mu[n] == trial_mu(n))
    check('trial_labels', (rad[n], support[n], single_sign[n]) == product_parameters(n))

    fs = factors(n)

    radn, sn, sign = product_parameters(n)
    actual, intersection = Counter(), Counter()

def mobius(n):
    sign = 1
    p = 2
    while p*p <= n:
        if n%p==0:
            n//=p
            sign=-sign
            if n%p==0:
                return 0
        p+=1
    return -sign if n>1 else sign

def ceildiv(a, b):
    return -((-a) // b)


class I:
    __slots__ = ("lo", "hi")

    def __init__(self, lo, hi=None):
        self.lo, self.hi = lo, lo if hi is None else hi
        assert self.lo <= self.hi

    @staticmethod
    def n(n):
        return I(n*S)

    @staticmethod
    def rat(a, b):
        assert b > 0
        return I(a*S//b, ceildiv(a*S, b))

    def __add__(self, other):
        return I(self.lo+other.lo, self.hi+other.hi)

    def __neg__(self):
        return I(-self.hi, -self.lo)

    def __sub__(self, other):
        return self + (-other)

    def scale(self, n):
        return I(self.lo*n, self.hi*n) if n >= 0 else I(self.hi*n, self.lo*n)

    def over(self, n):
        assert n > 0
        return I(self.lo//n, ceildiv(self.hi, n))

        return I(min(p)//S, ceildiv(max(p), S))

        return max(self.lo, other.lo) <= min(self.hi, other.hi)

    def sign(self):

        return {"lo": str(self.lo), "hi": str(self.hi), "scale": str(S)}

        p = p*z2
    tail = ceildiv(9*S, 4*(2*TERMS+1)*3**(2*TERMS+1))
    return I(2*total.lo, 2*total.hi+tail)


LOG2 = atanh_log(1, 3)


def logint(n):
    assert n >= 1
    k = n.bit_length()-1
    p = 1 << k
    return LOG2.scale(k) + atanh_log(n-p, n+p)


def mobius(n):
    mu = [1]*(n+1)
    mu[0] = 0
    prime = [True]*(n+1)
    for p in range(2, n+1):
        if prime[p]:
            for k in range(p, n+1, p):
                mu[k] = -mu[k]
                if k > p:
                    prime[k] = False
            for k in range(p*p, n+1, p*p):
                mu[k] = 0
    return mu


def deltas(mu, logs, last):
    a = b = I.n(0)
    out = {}
    for h in range(1, last+1):
        a = a + I.rat(mu[h], h)
        b = b + logs[h].scale(mu[h]).over(h)
        out[h] = I.n(1)-logs[h]*a+b
    return out


def window(h, mu, logs):
    pref = [I.n(0)]*(4*h)
    for k in range(h+1, 4*h):
        pref[k] = pref[k-1]+(logs[h]-logs[k]).scale(mu[k])
    diag = [I.n(0) for _ in range(3)]
    cross = [I.n(0) for _ in range(3)]
    total = I.n(0)
    for m in range(3*h, 4*h):
        v = [pref[m//j] if m//j > h else I.n(0) for j in (1, 2, 3)]
        total = total+isum(v).sq().over(m*(m+1))
        for j in range(3):
            diag[j] = diag[j]+v[j].sq().over(m*(m+1))
        for j, (a, b) in enumerate(((0, 1), (0, 2), (1, 2))):
            cross[j] = cross[j]+(v[a]*v[b]).scale(2).over(m*(m+1))
    assert total.overlap(isum(diag+cross))
    return {"J": total, "diag": diag, "cross": cross,
            "normalized": total/logs[h].sq()}


def coeffs(h, mu, logs):
    n = 6*((h*h+5)//6)
    k = n//6
    ell = logs[n]-logs[h]
    q = [I.n(0)]+[logs[n]-logs[max(h,j)] for j in range(1,n+1)]
    u = isum(q[j].scale(mu[j]).over(j) for j in range(1,n+1))
    v = ell.scale(2)+isum(q[j].scale(mu[j]) for j in range(1,n+1))
    s1 = isum(I.rat(1,j) for j in range(3*k+1,4*k+1))
    s2 = isum(I.rat(1,j) for j in range(5*k+1,6*k+1))
    a = (u-s2*v.over(k))/(s1-s2)
    z = (s1*v.over(k)-u)/(s1-s2)
    c = [I.n(0)]*(n+1)
    for j in range(2,n+1):
        correction = a if 3*k < j <= 4*k else z if 5*k < j <= 6*k else I.n(0)
        c[j] = (correction-q[j].scale(mu[j]))/ell
    assert isum(c[2:]).lo <= 3*S <= isum(c[2:]).hi
    moment = isum(c[j].over(j) for j in range(2,n+1))
    assert moment.lo <= S <= moment.hi
    logmoment = isum(c[j]*logs[j].over(j) for j in range(2,n+1))
    plateau = (I.n(1)-logmoment).sq().scale(h+1)
    delta = I(max(a.absolute().lo,z.absolute().lo),max(a.absolute().hi,z.absolute().hi))/ell
    tail_p6 = I.rat(14,3)/ell.sq()+delta.sq()*I.rat(5,3)
    t = math.isqrt(n**3)
    t += t*t < n**3
    l1 = isum(x.absolute() for x in c[2:])
    tail_elementary = (I.n(1)+l1).sq().over(t+1)
    return n,t,ell,a,z,c,plateau,tail_p6,tail_elementary


def main():

def rational_log(n):
    shift = n.bit_length()-1
    def series(a,b):
        z=F(a,b)
        p=z
        total=F(0)
        for j in range(80):
            total += 2*p/(2*j+1)
            p *= z*z
        remainder = 2*p/((161)*(1-z*z))
        return total,total+remainder
    a,b = series(n-(1<<shift),n+(1<<shift))
    l,u = series(1,3)
    a,b = a+shift*l,b+shift*u
    return I(a.numerator*S//a.denominator,ceildiv(b.numerator*S,b.denominator))


def independent_tables(limit):
    mu=[0]*(limit+1)
    mu[1]=1
    for d in range(1,limit+1):
        for n in range(2*d,limit+1,d):
            mu[n]-=mu[d]
    prime_logs={}
    logs=[I.n(0)]*(limit+1)
    for n in range(2,limit+1):
        z=n;p=2;value=I.n(0);omega=0;square=False
        while p*p<=z:
            if z%p==0:
                exponent=0
                while z%p==0:
                    z//=p;exponent+=1
                if p not in prime_logs:prime_logs[p]=rational_log(p)
                value=value+prime_logs[p].scale(exponent)
                omega+=1;square |= exponent>1
            p+=1
        if z>1:
            if z not in prime_logs:prime_logs[z]=rational_log(z)
            value=value+prime_logs[z];omega+=1
        logs[n]=value
        assert mu[n] == (0 if square else (-1)**omega)
        counts['independent_mu_trial_vs_inversion']+=1
    return mu,logs


def fromdata(x):
    assert int(x['scale'])==S
    return I(int(x['lo']),int(x['hi']))


def original_coefficients(h,n,mu,logs):
    ell=logs[n]-logs[h]
    q=[I.n(0)]+[logs[n]-logs[h] if j<=h else logs[n]-logs[j] for j in range(1,n+1)]
    u=isum(q[j].scale(mu[j]).over(j) for j in range(1,n+1))
    mass=ell.scale(2)+isum(q[j].scale(mu[j]) for j in range(1,n+1))
    k=n//6
    f1=sum((F(1,j) for j in range(3*k+1,4*k+1)),F(0))
    f2=sum((F(1,j) for j in range(5*k+1,n+1)),F(0))
    s1=I.rat(f1.numerator,f1.denominator)
    s2=I.rat(f2.numerator,f2.denominator)
    a=(u.scale(k)-mass*s2)/(s1-s2).scale(k)
    z=mass.over(k)-a
    c=[I.n(0)]*(n+1)
    for j in range(2,n+1):
        corr=a if 3*k<j<=4*k else z if 5*k<j<=n else I.n(0)
        c[j]=(corr-q[j].scale(mu[j]))/ell
    return c,ell,a,z


def literal_residue(m,c):
    lo=hi=0
    for k in range(2,len(c)):
        r=m%k
        lo+=c[k].lo*r//k
        hi+=ceildiv(c[k].hi*r,k)
    return I(S-hi,S-lo)


def raw_v(h,m,mu,logs):
    lo=hi=0
    for k in range(h+1,m+1):
        v=(logs[h]-logs[k]).scale(mu[k]*(m//k))
        lo+=v.lo;hi+=v.hi
    return I(lo,hi)


def raw_j(h,mu,logs):
    return isum(raw_v(h,m,mu,logs).sq().over(m*(m+1)) for m in range(3*h,4*h))


def transition(h,mu,logs):
    lam=logs[h+1]-logs[h]
    mixed=selfterm=enter=leave=I.n(0)
    for m in range(3*h+3,4*h):
        b=1-sum(mu[k]*(m//k) for k in range(1,h+1))
        v=raw_v(h,m,mu,logs)
        nxt=raw_v(h+1,m,mu,logs)
        assert nxt.overlap(v+lam.scale(b))
        counts['overlap_source_update']+=1
        mixed=mixed+(v*lam).scale(2*b).over(m*(m+1))
        selfterm=selfterm+lam.sq().scale(b*b).over(m*(m+1))
    for m in range(4*h,4*h+4):
        enter=enter+raw_v(h+1,m,mu,logs).sq().over(m*(m+1))
    for m in range(3*h,3*h+3):
        leave=leave+raw_v(h,m,mu,logs).sq().over(m*(m+1))
    return {'mixed':mixed,'self':selfterm,'boundary':enter-leave,
            'enter':enter,'leave':leave,'total':mixed+selfterm+enter-leave}

def main():
    start=time.perf_counter()

        variance=isum(isum(x[k] for k in range(d,n+1,d)).sq().scale(j2[d]) for d in range(2,n+1)).over(12)
